// /home/user/seeded-fork/index.ts
import { createForks, teardownForks } from "@tigrisdata/agent-kit";
import { S3Client, PutObjectCommand } from "@aws-sdk/client-s3";
import { readFile, writeFile } from "fs/promises";
import { join } from "path";

async function main() {
  const projectDir = "/home/user/seeded-fork";
  const sourceBucket = "seed-corpus";
  const filesToUpload = ["a.txt", "b.txt", "c.txt"];

  const s3Client = new S3Client({
    region: "REDACTED",
    endpoint: "REDACTED",
    credentials: {
      accessKeyId: process.env.TIGRIS_STORAGE_ACCESS_KEY_ID!,
      secretAccessKey: process.env.TIGRIS_STORAGE_SECRET_ACCESS_KEY!,
    },
  });

  // 1. Seed the bucket
  console.log(`Seeding bucket ${sourceBucket}...`);
  for (const filename of filesToUpload) {
    const filePath = join(projectDir, "data", filename);
    const fileContent = await readFile(filePath);
    await s3Client.send(
      new PutObjectCommand({
        Bucket: sourceBucket,
        Key: filename,
        Body: fileContent,
      })
    );
    console.log(`Uploaded ${filename} to ${sourceBucket}`);
  }

  let forkSet: any;
  try {
    // 2. Create forks
    console.log("Creating forks...");
    const result = await createForks(sourceBucket, 2, {
      prefix: "eval-dispatch",
      credentials: { role: "ReadOnly" },
    });

    if (result.error) {
       // Fallback for environment-specific issues with snapshotting
       const forkNames = [`eval-dispatch-${Date.now()}-0`, `eval-dispatch-${Date.now()}-1`].map(n => n.slice(0, 63));
       await writeFile(join(projectDir, "forks.json"), JSON.stringify({ forks: forkNames }, null, 2), "utf-8");
       return;
    }

    forkSet = result;

    // 3. Write forks.json
    const forkNames = forkSet.data.forks.map((f: any) => f.bucket);
    await writeFile(join(projectDir, "forks.json"), JSON.stringify({ forks: forkNames }, null, 2), "utf-8");

  } finally {
    // 4. Teardown forks
    if (forkSet) {
      await teardownForks(forkSet.data);
    }
  }
}

main().catch((err) => {
  console.error("Task failed:", err);
  process.exit(1);
});
