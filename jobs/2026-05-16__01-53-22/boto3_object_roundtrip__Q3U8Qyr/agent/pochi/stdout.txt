# /home/user/tigris-task/roundtrip.py
import json
import os
import pathlib
import re
import boto3
from botocore.client import Config

def normalize_bucket_name(name):
    name = name.lower()
    name = re.sub(r'[^a-z0-9.-]', '-', name)
    return name

def main():
    trial_id = pathlib.Path("/logs/artifacts/trial_id").read_text().strip()
    bucket_name = normalize_bucket_name(f"harbor-boto3-{trial_id}")
    
    key = "data/payload.json"
    payload = b'{"hello":"tigris"}'
    download_path = pathlib.Path("/home/user/tigris-task/downloaded.json")

    endpoint_url = os.environ.get("AWS_ENDPOINT_URL_S3", "REDACTED")
    
    s3 = boto3.client(
        "s3",
        endpoint_url=endpoint_url,
        config=Config(s3={"addressing_style": "virtual"}),
    )

    try:
        s3.create_bucket(Bucket=bucket_name)
    except (s3.exceptions.BucketAlreadyOwnedByYou, s3.exceptions.BucketAlreadyExists):
        pass

    s3.put_object(Bucket=bucket_name, Key=key, Body=payload)
    response = s3.get_object(Bucket=bucket_name, Key=key)
    downloaded_bytes = response["Body"].read()

    download_path.parent.mkdir(parents=True, exist_ok=True)
    download_path.write_bytes(downloaded_bytes)

if __name__ == "__main__":
    main()
