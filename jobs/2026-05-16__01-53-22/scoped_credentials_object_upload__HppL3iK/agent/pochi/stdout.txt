```typescript
import { createWorkspace, teardownWorkspace } from "@tigrisdata/agent-kit";
import { S3Client, PutObjectCommand } from "@aws-sdk/client-s3";

async function main() {
  let workspaceResponse;
  try {
    workspaceResponse = await createWorkspace("scoped-upload-ws", {
      credentials: { role: "Editor" },
    });

    if (workspaceResponse.error) {
      throw workspaceResponse.error;
    }

    const workspace = workspaceResponse.data;

    if (!workspace.bucket || !workspace.credentials?.accessKeyId || !workspace.credentials?.secretAccessKey) {
      throw new Error("Workspace or scoped credentials not properly provisioned");
    }

    const s3Client = new S3Client({
      region: "REDACTED",
      endpoint: "REDACTED",
      credentials: {
        accessKeyId: workspace.credentials.accessKeyId,
        secretAccessKey: workspace.credentials.secretAccessKey,
      },
    });

    await s3Client.send(
      new PutObjectCommand({
        Bucket: workspace.bucket,
        Key: "greeting.txt",
        Body: "hello scoped",
      })
    );

    console.log(`SCOPED_UPLOAD_OK ${workspace.bucket}`);
  } catch (error) {
    console.error("Error during scoped upload:", error);
    process.exit(1);
  } finally {
    if (workspaceResponse?.data) {
      await teardownWorkspace(workspaceResponse.data);
    }
  }
}

main();
```
