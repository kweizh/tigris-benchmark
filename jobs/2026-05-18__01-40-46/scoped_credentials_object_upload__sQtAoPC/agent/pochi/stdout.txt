import { createWorkspace, teardownWorkspace } from "@tigrisdata/agent-kit";
import { S3Client, PutObjectCommand } from "@aws-sdk/client-s3";
import * as fs from "fs";

async function main() {
  let workspace: any;
  try {
    const trialId = fs.readFileSync("/logs/artifacts/trial_id", "utf-8").trim();
    const workspaceName = `harbor-scoped-${trialId}`
      .toLowerCase()
      .replace(/[^a-z0-9.-]/g, "-");

    const response = await createWorkspace(workspaceName, {
      credentials: { role: "Editor" },
    });

    if (response.error) {
      throw response.error;
    }

    workspace = response.data;

    if (!workspace.bucket || !workspace.credentials) {
      throw new Error("Workspace bucket or credentials missing in response");
    }

    const s3Client = new S3Client({
      region: "REDACTED",
      endpoint: "REDACTED",
      credentials: {
        accessKeyId: workspace.credentials.accessKeyId,
        secretAccessKey: workspace.credentials.secretAccessKey,
      },
    });

    await s3Client.send(
      new PutObjectCommand({
        Bucket: workspace.bucket,
        Key: "greeting.txt",
        Body: "hello scoped",
      })
    );

    console.log(`SCOPED_UPLOAD_OK ${workspace.bucket}`);
  } catch (error) {
    console.error(error);
    process.exit(1);
  } finally {
    if (workspace) {
      await teardownWorkspace(workspace);
    }
  }
}

main();
